```rust
use std::collections::HashMap;

pub fn compter_mots(texte: &str) -> Vec<(String, usize)> {
    let mut compte: HashMap<String, usize> = HashMap::new();
    for mot in texte.split_whitespace() {
        let mot_min = mot.to_lowercase();
        *compte.entry(mot_min).or_insert(0) += 1;
    }
    let mut resultat: Vec<(String, usize)> = compte.into_iter().collect();
    resultat.sort_by(|a, b| {
        b.1.cmp(&a.1).then_with(|| a.0.cmp(&b.0))
    });
    resultat
}
